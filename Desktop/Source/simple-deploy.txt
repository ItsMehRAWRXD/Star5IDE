# Simple Manual Deployment Steps
# RawrZ Security Platform to lionfish-app-ey7kn.ondigitalocean.app/panel

## Step 1: Get your droplet IP
Run this command in your terminal:
doctl compute droplet list

## Step 2: Copy files to your server
Replace YOUR_SERVER_IP with the IP from step 1:

scp Dockerfile.prod root@YOUR_SERVER_IP:/root/rawrz-panel/
scp docker-compose.panel.yml root@YOUR_SERVER_IP:/root/rawrz-panel/
scp env.panel.example root@YOUR_SERVER_IP:/root/rawrz-panel/
scp -r src/ root@YOUR_SERVER_IP:/root/rawrz-panel/
scp -r public/ root@YOUR_SERVER_IP:/root/rawrz-panel/
scp -r scripts/ root@YOUR_SERVER_IP:/root/rawrz-panel/
scp package*.json root@YOUR_SERVER_IP:/root/rawrz-panel/

## Step 3: SSH into your server
ssh root@YOUR_SERVER_IP

## Step 4: Deploy on server
cd /root/rawrz-panel
apt-get update
apt-get install -y nginx docker.io docker-compose
cp env.panel.example .env.production
docker-compose -f docker-compose.panel.yml up -d --build

## Step 5: Setup nginx
cp nginx/panel-nginx.conf /etc/nginx/sites-available/rawrz-panel
ln -sf /etc/nginx/sites-available/rawrz-panel /etc/nginx/sites-enabled/
rm -f /etc/nginx/sites-enabled/default
nginx -t
systemctl restart nginx

## Step 6: Test
Visit: https://lionfish-app-ey7kn.ondigitalocean.app/panel
Health check: https://lionfish-app-ey7kn.ondigitalocean.app/panel/health
